[General]

# 建议开启，避免系统服务/推送兼容问题

# ---------- 本机 / 局域网 ----------

# ---------- TUN 旁路 ----------
# Shadowrocket 正确参数名称为 tun-excluded-routes

# ---------- DNS ----------
# 速度/简单优先，继续使用系统 DNS
dns-server = system
fallback-dns-server = system


# ---------- IPv6 ----------
# 先关闭，减少不同节点/网络之间的兼容变量
ipv6 = false
prefer-ipv6 = false


# ---------- 私有 IP ----------
private-ip-answer = true


# ---------- UDP / QUIC ----------
# 节点 UDP 支持良好时允许 QUIC / HTTP3
block-quic = always-allow

# 节点不支持 UDP 时直接拒绝，不让流量意外直连
udp-policy-not-supported-behaviour = REJECT
